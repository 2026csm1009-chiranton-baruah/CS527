% Fibonacci sequence - first 10 terms
% Results stored at addresses 200..236
% Temporary storage:
%   240 = old x2
%   244 = new Fibonacci value

x1 = 0
x2 = 1
x3 = 200
x4 = 0

.loop
[x3] = x1

% Save current x2
x6 = 240
[x6] = x2

% Calculate next Fibonacci number
x5 = x1 + x2

% x1 = old x2
x6 = 240
x1 = [x6]

% x2 = new Fibonacci number
x6 = 244
[x6] = x5
x2 = [x6]

% Advance result pointer and counter
x3 = x3 + 4
x4 = x4 + 1

x6 = x4 - 10
BEQ .exit
BAL .loop

.exit
