% ============================================================
% NON-RESTORING DIVISION
%
% 16-bit unsigned dividend / divisor
%
% A = partial remainder
% Q = dividend, then quotient
% M = divisor
%
% ============================================================

% ============================================================
% CONSTANTS
% ============================================================

% Build 65536
x8 = 1
x8 = x8 << 16

% Build 65535 = 0xFFFF
x7 = x8 - 1

% ============================================================
% TEST 1: 13 / 3
% ============================================================

x1 = 3
x2 = 13
x3 = 0
x4 = 16

.t1loop

% ------------------------------------------------------------
% Save sign of A BEFORE modifying A or Q.
% ------------------------------------------------------------

x6 = x3 + 0
BGE .t1positive

% ------------------------------------------------------------
% A < 0
%
% First shift Q and extract its MSB.
% Then:
%       A = (A << 1) + bit + M
% ------------------------------------------------------------

x5 = x2 >> 15

x2 = x2 << 1

% Keep Q to 16 bits
x6 = x2 >> 16
x6 = x6 << 16
x2 = x2 - x6

x3 = x3 << 1
x3 = x3 + x5
x3 = x3 + x1

BAL .t1quotient

.t1positive

% ------------------------------------------------------------
% A >= 0
%
% First shift Q and extract its MSB.
% Then:
%       A = (A << 1) + bit - M
% ------------------------------------------------------------

x5 = x2 >> 15

x2 = x2 << 1

% Keep Q to 16 bits
x6 = x2 >> 16
x6 = x6 << 16
x2 = x2 - x6

x3 = x3 << 1
x3 = x3 + x5
x3 = x3 - x1

% ------------------------------------------------------------
% Set quotient bit from NEW A.
% ------------------------------------------------------------

.t1quotient

BLT .t1next

% New A >= 0 -> quotient bit = 1
x2 = x2 + 1

.t1next

x4 = x4 - 1
BNE .t1loop

% ------------------------------------------------------------
% Final correction
%
% If A < 0, A = A + M
% ------------------------------------------------------------

x6 = x3 + 0
BLT .t1final
BAL .t1print

.t1final
x3 = x3 + x1

.t1print

Print x2
Print x3
