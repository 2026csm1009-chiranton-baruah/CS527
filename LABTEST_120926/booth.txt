% ============================================================
% Booth Multiplication
% Signed 16-bit x Signed 16-bit
%
% Memory:
%   [48] = Multiplicand M
%   [40] = Multiplier Q
%
% Registers:
%   x1  = M
%   x2  = Q
%   x3  = A
%   x4  = Q-1
%   x5  = iteration counter
%   x6  = Q0
%   x7  = temporary
%   x8  = temporary
%   x9  = A0
%   x11 = 32768 (bit 15)
%   x13 = final product
%
% Booth rules:
%
%   Q0 Q-1
%    0   0   -> Nothing
%    0   1   -> A = A + M
%    1   0   -> A = A - M
%    1   1   -> Nothing
%
% ============================================================


% ------------------------------------------------------------
% Load operands
% ------------------------------------------------------------

x1 = [48]
x2 = [40]


% ------------------------------------------------------------
% Initialize accumulator and Q-1
% ------------------------------------------------------------

x3 = 0
x4 = 0


% ------------------------------------------------------------
% Counter = 16
% ------------------------------------------------------------

x5 = 16


% ------------------------------------------------------------
% Construct 32768 = 1 << 15
% This is bit 15 of the 16-bit Q register.
% ------------------------------------------------------------

x11 = 1
x11 = x11 << 15


% ============================================================
% Booth loop
% ============================================================

.loop


% ------------------------------------------------------------
% Extract Q0
%
% Q0 = Q - ((Q >> 1) << 1)
% ------------------------------------------------------------

x7 = x2 >> 1
x8 = x7 << 1
x6 = x2 - x8


% ------------------------------------------------------------
% Booth decision
%
% Q0 = 0, Q-1 = 0 -> nothing
% Q0 = 0, Q-1 = 1 -> A = A + M
% Q0 = 1, Q-1 = 0 -> A = A - M
% Q0 = 1, Q-1 = 1 -> nothing
% ------------------------------------------------------------


% Test Q0
x7 = x6 + 0
BEQ .qzero


% ------------------------------------------------------------
% Q0 = 1
% ------------------------------------------------------------

% Test Q-1
x7 = x4 + 0
BEQ .subm


% Q0 = 1, Q-1 = 1
% Nothing
BAL .savebit


% ------------------------------------------------------------
% Q0 = 0
% ------------------------------------------------------------

.qzero

% Test Q-1
x7 = x4 + 0
BNE .addm


% Q0 = 0, Q-1 = 0
% Nothing
BAL .savebit


% ------------------------------------------------------------
% A = A + M
% ------------------------------------------------------------

.addm

x3 = x3 + x1
BAL .savebit


% ------------------------------------------------------------
% A = A - M
% ------------------------------------------------------------

.subm

x3 = x3 - x1


% ============================================================
% Save Q0 as Q-1
% ============================================================

.savebit

x4 = x6 + 0


% ------------------------------------------------------------
% Extract A0 before shifting
%
% A0 = A - ((A >> 1) << 1)
% ------------------------------------------------------------

x7 = x3 >> 1
x8 = x7 << 1
x9 = x3 - x8


% ------------------------------------------------------------
% Arithmetic right shift A
% Logical right shift Q
% ------------------------------------------------------------

x3 = x3 >>> 1
x2 = x2 >> 1


% ------------------------------------------------------------
% Insert old A0 into Q bit 15
%
% If A0 == 1:
%     Q = Q + 32768
% ------------------------------------------------------------

x7 = x9 - 1
BEQ .insertmsb

BAL .next


.insertmsb

x2 = x2 + x11


% ============================================================
% Next Booth iteration
% ============================================================

.next

x5 = x5 - 1
BNE .loop


% ============================================================
% Construct final product
%
% Product = (A << 16) + Q
% ============================================================

x3 = x3 << 16
x13 = x3 + x2


% ------------------------------------------------------------
% Print result
% ------------------------------------------------------------

Print x13
