% ============================================================
% Bitwise parity algorithm
%
% Computes the parity bit of an 8-bit integer using the new
% logical operators AND, XOR and shifts.
%
% Algorithm:
%   x = x ^ (x >> 4)
%   x = x ^ (x >> 2)
%   x = x ^ (x >> 1)
%   parity = x & 1
%
% Result:
%   0 = even number of set bits
%   1 = odd number of set bits
%
% Test values:
%   0   -> 0 set bits -> 0
%   1   -> 1 set bit  -> 1
%   3   -> 2 set bits -> 0
%   7   -> 3 set bits -> 1
%   15  -> 4 set bits -> 0
%   170 -> 4 set bits -> 0
%   171 -> 5 set bits -> 1
%   255 -> 8 set bits -> 0
% ============================================================

% ------------------------------------------------------------
% Parity of 0
% ------------------------------------------------------------

x1 = 0
x2 = x1 >> 4
x1 = x1 ^ x2
x2 = x1 >> 2
x1 = x1 ^ x2
x2 = x1 >> 1
x1 = x1 ^ x2
x3 = x1 & 1
Print x3
