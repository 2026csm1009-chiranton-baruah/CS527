% ============================================================
% Recurring_division.txt
%
% 16-bit unsigned restoring binary division
%
% Algorithm:
%
%   A = 0
%   Q = dividend
%   M = divisor
%
%   Repeat 16 times:
%
%       bit = Q[15]
%       Q = Q << 1
%       Q = Q & 0xFFFF
%
%       A = (A << 1) + bit
%
%       A = A - M
%
%       if A < 0:
%           A = A + M
%           Q[0] = 0
%       else:
%           Q[0] = 1
%
%   Final:
%       Q = quotient
%       A = remainder
%
% Tests:
%   13  / 3   = 4  remainder 1
%   11  / 2   = 5  remainder 1
%   100 / 10  = 10 remainder 0
%   255 / 7   = 36 remainder 3
%
% ============================================================


% ============================================================
% CONSTANTS
% ============================================================

% Build 65536
x8 = 1
x8 = x8 << 16

% Build 65535 = 0xFFFF
x7 = x8 - 1


% ============================================================
% TEST 1: 13 / 3
% ============================================================

x1 = 3
x2 = 13

% A = 0
x3 = 0

% Counter = 16
x4 = 16

.t1loop

% ------------------------------------------------------------
% Extract Q[15]
% ------------------------------------------------------------
x5 = x2 >> 15

% ------------------------------------------------------------
% Q = Q << 1
% ------------------------------------------------------------
x2 = x2 << 1

% ------------------------------------------------------------
% Keep only lower 16 bits of Q
%
% x6 = upper 16 bits
% x6 = x6 << 16
% x2 = x2 - x6
% ------------------------------------------------------------
x6 = x2 >> 16
x6 = x6 << 16
x2 = x2 - x6

% ------------------------------------------------------------
% A = (A << 1) + extracted bit
% ------------------------------------------------------------
x3 = x3 << 1
x3 = x3 + x5

% ------------------------------------------------------------
% A = A - M
% ------------------------------------------------------------
x6 = x3 - x1
BLT .t1restore

% ------------------------------------------------------------
% Successful subtraction
% Quotient bit = 1
% ------------------------------------------------------------
x3 = x6 + 0
x2 = x2 + 1
BAL .t1next

.t1restore

% ------------------------------------------------------------
% Failed subtraction
% Restore A
% Quotient bit = 0
% ------------------------------------------------------------
x3 = x6 + x1

.t1next

x4 = x4 - 1
BNE .t1loop

Print x2
Print x3


% ============================================================
% TEST 2: 11 / 2
% ============================================================

x1 = 2
x2 = 11
x3 = 0
x4 = 16

.t2loop

x5 = x2 >> 15

x2 = x2 << 1

% Mask Q to 16 bits
x6 = x2 >> 16
x6 = x6 << 16
x2 = x2 - x6

% A = (A << 1) + Q[15]
x3 = x3 << 1
x3 = x3 + x5

% A = A - M
x6 = x3 - x1
BLT .t2restore

% Successful subtraction
x3 = x6 + 0
x2 = x2 + 1
BAL .t2next

.t2restore

% Restore A
x3 = x6 + x1

.t2next

x4 = x4 - 1
BNE .t2loop

Print x2
Print x3
