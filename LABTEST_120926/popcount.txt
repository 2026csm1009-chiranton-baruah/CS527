% Bitwise population-count test
%
% Counts the number of set bits in each 8-bit value.
% Uses the new logical operators together with shifts,
% arithmetic, and a loop:
%
%   count = count + (x & 1)
%   x = x >> 1
%
% The algorithm exercises AND, ANDI, SRLI, ADDI, branches,
% and register assignments.
% ============================================================

% ------------------------------------------------------------
% Test 0: 00000000 -> 0
% ------------------------------------------------------------

.L0
x4 = x1 & x3
x2 = x2 + x4

x5 = x1 - 0
BLE .D0
BAL .L0

.D0
